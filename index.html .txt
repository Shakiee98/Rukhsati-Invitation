      // 1. Reveal main content in DOM
      mainInvitation.classList.add('visible');

      // 2. Animate cover transition
      coverScreen.classList.add('unlocked');

      // 3. Trigger initial scroll position smoothly & initialize components
      setTimeout(() => {
        window.scrollTo({ top: 0, behavior: 'smooth' });
        initScratchCard();
        initScrollObservers();
      }, 300);
    });

    /* -------------------------------------------------------------
       TOUCH SCRATCH CARD (HTML5 Canvas)
       Conceals date until >=30% has been cleared, then gracefully fades.
       ------------------------------------------------------------- */
    function initScratchCard() {
      const container = document.getElementById('scratchCardContainer');
      const canvas = document.getElementById('scratchCanvas');
      const hint = document.getElementById('scratchPrompt');
      const ctx = canvas.getContext('2d', { willReadFrequently: true });

      const width = container.offsetWidth;
      const height = container.offsetHeight;

      // Handle retina displays
      const dpr = window.devicePixelRatio || 1;
      canvas.width = width * dpr;
      canvas.height = height * dpr;
      ctx.scale(dpr, dpr);

      // Draw Luxury Champagne Gold Foil Cover
      function renderFoilSurface() {
        const grad = ctx.createLinearGradient(0, 0, width, height);
        grad.addColorStop(0, '#ebd9b4');
        grad.addColorStop(0.3, '#d4bc8b');
        grad.addColorStop(0.7, '#ba9f67');
        grad.addColorStop(1, '#a68a52');
        ctx.fillStyle = grad;
        ctx.fillRect(0, 0, width, height);

        // Subtle decorative border on foil
        ctx.strokeStyle = 'rgba(255, 255, 255, 0.4)';
        ctx.lineWidth = 1;
        ctx.strokeRect(8, 8, width - 16, height - 16);

        // Center typography on scratch layer
        ctx.font = '500 11px "Montserrat", sans-serif';
        ctx.fillStyle = '#4a3d24';
        ctx.textAlign = 'center';
        ctx.textBaseline = 'middle';
        ctx.letterSpacing = '3px';
        ctx.fillText('SCRATCH TO REVEAL', width / 2, height / 2);
      }

      renderFoilSurface();

      let isDrawing = false;
      let revealed = false;
      let erasedPixelsCount = 0;

      function getPosition(e) {
        const rect = canvas.getBoundingClientRect();
        let clientX, clientY;
        if (e.touches && e.touches.length > 0) {
          clientX = e.touches[0].clientX;
          clientY = e.touches[0].clientY;
        } else {
          clientX = e.clientX;
          clientY = e.clientY;
        }
        return {
          x: clientX - rect.left,
          y: clientY - rect.top
        };
      }

      function scratch(e) {
        if (!isDrawing || revealed) return;
        e.preventDefault();

        const { x, y } = getPosition(e);
        ctx.globalCompositeOperation = 'destination-out';
        ctx.beginPath();
        ctx.arc(x, y, 22, 0, Math.PI * 2, false);
        ctx.fill();

        checkPercentageScratched();
      }

      function checkPercentageScratched() {
        if (revealed) return;

        // Sample every 16th pixel for high-performance touch response
        const sampleGap = 16;
        const imgData = ctx.getImageData(0, 0, canvas.width, canvas.height);
        const data = imgData.data;
        let transparentPixels = 0;
        let totalSamples = 0;

        for (let i = 3; i < data.length; i += 4 * sampleGap) {
          totalSamples++;
          if (data[i] === 0) {
            transparentPixels++;
          }
        }

        const scratchedPercentage = (transparentPixels / totalSamples) * 100;

        // Reveal threshold set at 30%
        if (scratchedPercentage >= 30) {
          revealed = true;
          canvas.style.opacity = '0';
          canvas.style.pointerEvents = 'none';
          hint.textContent = "Date Unlocked";
          hint.style.color = "var(--gold-deep)";
          hint.style.letterSpacing = "0.25em";
        }
      }

      // Touch & Mouse Listeners
      canvas.addEventListener('mousedown', (e) => { isDrawing = true; scratch(e); });
      window.addEventListener('mouseup', () => { isDrawing = false; });
      canvas.addEventListener('mousemove', scratch);

      canvas.addEventListener('touchstart', (e) => { isDrawing = true; scratch(e); }, { passive: false });
      window.addEventListener('touchend', () => { isDrawing = false; });
      canvas.addEventListener('touchmove', scratch, { passive: false });
    }

    /* -------------------------------------------------------------
       SCROLL OBSERVER FOR DELICATE REVEAL ANIMATION
       ------------------------------------------------------------- */
    function initScrollObservers() {
      const observer = new IntersectionObserver((entries) => {
        entries.forEach(entry => {
          if (entry.isIntersecting) {
            entry.target.classList.add('visible');
          }
        });
      }, {
        threshold: 0.15
      });

      document.querySelectorAll('.fade-up').forEach(el => {
        observer.observe(el);
      });
    }
  </script>
